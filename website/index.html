# Inside MyIngress — source excerpt
@action
def ipv4_forward(self, dstAddr: bit48, port: bit9) -> None:
    self.assign(self.meta.egress_port, port)
    self.assign(self.hdr.ethernet.srcAddr, self.hdr.ethernet.dstAddr)
    self.assign(self.hdr.ethernet.dstAddr, dstAddr)
    self.assign(self.hdr.ipv4.ttl, self.hdr.ipv4.ttl - 1)

def forwardAction : Action :=
  ⟨"ipv4_forward", [⟨"dstAddr", .bits 48, .none⟩, ⟨"port", .bits 9, .none⟩], [
    .assign egress.lvalue (.var "port"),
    .assign ethSrc.lvalue ethDst.expr,
    .assign ethDst.lvalue (.var "dstAddr"),
    .assign ttl.lvalue (.binary .sub ttl.expr (.literal (.bits 8 1))) ]⟩
